def payload(url, user, password):
    request_url = url
    username = user
    password = password
    encoded_credentials = b64encode(bytes(f'{username}:{password}',
    encoding='ascii')).decode('ascii')
    auth_header = f'Basic {encoded_credentials}'

    # payload = '{"kind":"vm"}'
    payload = '{"kind":"vm","length":500}'

    headers = {'Accept': 'application/json', 'Content-Type': 'application/json',
    'Authorization': f'{auth_header}', 'cache-control': 'no-cache'}

    response = requests.request('post', request_url, data=payload, headers=headers) # Verify = false was in here
    print(response.status_code)
    # Is diary cut off?
    diary = response.json()
    return diary
